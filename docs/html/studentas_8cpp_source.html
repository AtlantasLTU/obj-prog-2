22double Studentas::galutinis(bool medianos, int ndKiekis) const
23{
24 const int max = (ndKiekis != 0) ? ndKiekis : maxNdKiekis;
25 if(nd.empty()) return 0.6*rez;
26 if(medianos)
27 {
28 std::vector<int> ND = nd;
29 if(ND.size() < max) ND.resize(max, 0);
30 std::sort(ND.begin(), ND.end());
31 const int vidurys = max / 2;
32 std::nth_element(ND.begin(), ND.begin() + vidurys, ND.end());
33 double med;
34 if(max % 2 == 0)
35 {
36 med = (*std::max_element(ND.begin(), ND.begin() + vidurys) + ND[vidurys]) / 2.0;
37 }
38 else
39 {
40 med = ND[vidurys];
41 }
42 return 0.4 * med + 0.6 * rez;
43 }
44 else
45 {
46 int sum = std::accumulate(nd.begin(), nd.end(), 0);
47
48 double vid = (double)sum / max;
49 return 0.4 * vid + 0.6 * rez;
50 }
51}
